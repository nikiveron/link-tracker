# PromQL запросы для дашбордов LinkTracker Monitoring

================================================================================
# RED METRICS DASHBOARD
================================================================================

# 1. Rate (Requests per second)
# Описание: Количество запросов в секунду, которые обрабатывает приложение
# Визуализация: Line chart
rate(http_requests_total{app="$app_name"}[5m])

# 2. Errors (5xx per second)
# Описание: Количество ошибочных запросов (5xx) в секунду
# Визуализация: Line chart
rate(http_requests_total{app="$app_name", status=~"5.."}[5m])

# 3. Duration - p50 (50-й перцентиль)
# Описание: 50% запросов обрабатываются быстрее этого значения
# Визуализация: Line chart
histogram_quantile(0.50, rate(http_request_duration_seconds_bucket{app="$app_name"}[5m]))

# 4. Duration - p95 (95-й перцентиль)
# Описание: 95% запросов обрабатываются быстрее этого значения
# Визуализация: Line chart
histogram_quantile(0.95, rate(http_request_duration_seconds_bucket{app="$app_name"}[5m]))

# 5. Duration - p99 (99-й перцентиль)
# Описание: 99% запросов обрабатываются быстрее этого значения
# Визуализация: Line chart
histogram_quantile(0.99, rate(http_request_duration_seconds_bucket{app="$app_name"}[5m]))

# 6. Memory Usage - Working Set
# Описание: Рабочий набор памяти приложения (RAM)
# Визуализация: Line chart
process_working_set_bytes{app="$app_name"}

# 7. Memory Usage - Private Memory
# Описание: Частная память приложения (без общих страниц)
# Визуализация: Line chart
process_private_memory_bytes{app="$app_name"}

# Описание: Количество пользовательских запросов к приложению в секунду
# Визуализация: Line chart
# Лейблы: app (scrapper, bot, ai-agent)
rate(http_requests_total{app="$app_type"}[5m])

# Визуализация: Line chart

# 13. Количество отправленных нотификаций

# Визуализация: Line chart
# Лейблы: app=bot
rate(sent_notification_total{app="bot"}[5m])

================================================================================
# ДОПОЛНИТЕЛЬНЫЕ ЗАПРОСЫ ДЛЯ АНАЛИЗА
================================================================================

# 14. Общее количество запросов за последний час
# Описание: Суммарное количество HTTP запросов за последний час
sum(increase(http_requests_total{app="$app_name"}[1h]))

# 15. Процент ошибок (Error Rate)
# Описание: Доля ошибочных запросов (5xx) от общего количества в процентах
rate(http_requests_total{app="$app_name", status=~"5.."}[5m]) 
/ 
rate(http_requests_total{app="$app_name"}[5m]) 
* 100

# 16. Доступность сервиса (Availability)
# Описание: Процент успешных запросов (не 5xx) от общего количества
1 - (
  rate(http_requests_total{app="$app_name", status=~"5.."}[5m]) 
  / 
  rate(http_requests_total{app="$app_name"}[5m])
)

# 17. Среднее время ответа
# Описание: Среднее время обработки запроса в секундах
rate(http_request_duration_seconds_sum{app="$app_name"}[5m]) 
/ 
rate(http_request_duration_seconds_count{app="$app_name"}[5m])

# 18. Количество ссылок на мониторинге (по доменам)
# Описание: Детализация количества отслеживаемых ссылок по доменам
links_on_track_total{app="scrapper"} by (tracked_source)

# 19. Топ команд бота по частоте использования
# Описание: Рейтинг команд бота по количеству вызовов за последний час
topk(5, sum(increase(command_requests_total{app="bot"}[1h])) by (command))

# 20. Скорость потребления памяти
# Описание: Скорость изменения использования памяти (байт/сек)
deriv(process_working_set_bytes{app="$app_name"}[5m])

================================================================================
# ЗАПРОСЫ ДЛЯ АЛЕРТОВ (Grafana Alerting)
================================================================================

# 21. Алерт на высокое потребление RAM
# Описание: Срабатывает, когда потребление RAM превышает 500MB
# Threshold: > 524288000 (500MB)
process_working_set_bytes{app="$app_name"} > 524288000

# 22. Алерт на высокий Error Rate
# Описание: Срабатывает, когда доля ошибок превышает 5%
rate(http_requests_total{app="$app_name", status=~"5.."}[5m]) 
/ 
rate(http_requests_total{app="$app_name"}[5m]) 
> 0.05

# 23. Алерт на высокую задержку (p95)
# Описание: Срабатывает, когда 95-й перцентиль превышает 2 секунды
histogram_quantile(0.95, rate(http_request_duration_seconds_bucket{app="$app_name"}[5m])) > 2

# 24. Алерт на недоступность сервиса
# Описание: Срабатывает, когда сервис не отвечает (down)
up{job="$app_name"} == 0

# 25. Алерт на отсутствие уведомлений
# Описание: Срабатывает, если бот не отправлял уведомления более 30 минут
increase(sent_notification_total{app="bot"}[30m]) == 0
